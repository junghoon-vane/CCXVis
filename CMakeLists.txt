cmake_minimum_required(VERSION 3.20)
project(CCXVis VERSION 0.1.0 LANGUAGES C CXX)

include(CTest)
include(GNUInstallDirs)

# Seed renamed settings from existing build caches and command lines.
function(ccxvis_cache_setting name type description default)
    set(legacy_name "CGX_${name}")
    if(DEFINED ${legacy_name} AND NOT DEFINED CCXVIS_${name})
        set(default "${${legacy_name}}")
    endif()
    set(CCXVIS_${name} "${default}" CACHE ${type} "${description}")
endfunction()

ccxvis_cache_setting(BUILD_GUI BOOL "Build the desktop viewer" ON)
ccxvis_cache_setting(TEST_RENDERING BOOL "Run the PNG smoke test on an OpenGL-capable desktop" OFF)
ccxvis_cache_setting(QT_MAJOR STRING "Qt major version used by the VTK build" 6)
ccxvis_cache_setting(PARAVIEW_VTK_ROOT PATH "Use the VTK headers and shared libraries in the Rocky Linux ParaView RPMs" "")
ccxvis_cache_setting(BASELINE_FRD FILEPATH "Optional upstream CGX 2.23 examples/result.frd" "")

add_library(ccxvis_frd STATIC src/FrdDocument.cpp src/ContourSequence.cpp)
target_compile_features(ccxvis_frd PUBLIC cxx_std_17)
target_include_directories(ccxvis_frd PUBLIC src)

if(CCXVIS_BUILD_GUI)
    find_package(Qt${CCXVIS_QT_MAJOR} REQUIRED COMPONENTS Widgets Concurrent)
    if(CCXVIS_QT_MAJOR STREQUAL "6")
        find_package(Qt6 REQUIRED COMPONENTS OpenGLWidgets)
    endif()
    set(ccxvis_vtk_modules
        CommonCore CommonDataModel CommonExecutionModel FiltersCore FiltersGeneral FiltersGeometry GUISupportQt
        InteractionStyle InteractionWidgets RenderingAnnotation RenderingCore RenderingFreeType RenderingOpenGL2)
    if(CCXVIS_PARAVIEW_VTK_ROOT)
        if(NOT CCXVIS_QT_MAJOR STREQUAL "5")
            message(FATAL_ERROR "The Rocky ParaView RPM build uses Qt 5")
        endif()
        add_library(ccxvis_paraview_vtk INTERFACE)
        find_path(ccxvis_vtk_include QVTKOpenGLNativeWidget.h
            PATHS "${CCXVIS_PARAVIEW_VTK_ROOT}/include/openmpi-x86_64/paraview/paraview" REQUIRED NO_DEFAULT_PATH)
        target_include_directories(ccxvis_paraview_vtk SYSTEM INTERFACE "${ccxvis_vtk_include}")
        foreach(module IN LISTS ccxvis_vtk_modules ITEMS sys)
            find_library(ccxvis_vtk_${module} vtk${module}
                PATHS "${CCXVIS_PARAVIEW_VTK_ROOT}/lib64/openmpi/lib/paraview" REQUIRED NO_DEFAULT_PATH)
            target_link_libraries(ccxvis_paraview_vtk INTERFACE "${ccxvis_vtk_${module}}")
        endforeach()
        set(vtk_libraries ccxvis_paraview_vtk)
    else()
        find_package(VTK REQUIRED COMPONENTS ${ccxvis_vtk_modules})
        get_target_property(vtk_qt_targets VTK::GUISupportQt INTERFACE_LINK_LIBRARIES)
        if(vtk_qt_targets MATCHES "Qt[56]::" AND NOT vtk_qt_targets MATCHES "Qt${CCXVIS_QT_MAJOR}::")
            message(FATAL_ERROR "VTK and CCXVIS_QT_MAJOR use different Qt versions")
        endif()
        set(vtk_libraries ${VTK_LIBRARIES})
    endif()
    add_library(ccxvis_ui STATIC src/MainWindow.cpp src/Viewport.cpp src/MovieExport.cpp
        src/CurveData.cpp src/CurvePlot.cpp src/MainWindowAnimation.cpp)
    target_link_libraries(ccxvis_ui PUBLIC ccxvis_frd
        Qt${CCXVIS_QT_MAJOR}::Widgets Qt${CCXVIS_QT_MAJOR}::Concurrent ${vtk_libraries})
    add_executable(ccxvis src/main.cpp)
    target_link_libraries(ccxvis PRIVATE ccxvis_ui)
    if(CCXVIS_PARAVIEW_VTK_ROOT)
        target_compile_definitions(ccxvis_ui PRIVATE CCXVIS_EXPLICIT_VTK_AUTOINIT)
        set_target_properties(ccxvis PROPERTIES INSTALL_RPATH
            "${CCXVIS_PARAVIEW_VTK_ROOT}/lib64/openmpi/lib/paraview")
    else()
        vtk_module_autoinit(TARGETS ccxvis_ui MODULES ${vtk_libraries})
    endif()
    install(TARGETS ccxvis RUNTIME DESTINATION "${CMAKE_INSTALL_BINDIR}")
    install(FILES LICENSE DESTINATION "${CMAKE_INSTALL_DATAROOTDIR}/licenses/ccxvis")
endif()
